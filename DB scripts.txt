        WHEN 4 THEN 'Daily'
        WHEN 8 THEN 'Weekly'
        WHEN 16 THEN 'Monthly'

    STUFF(STUFF(RIGHT('000000' + CAST(s.active_start_time AS VARCHAR(6)), 6), 5, 0, ':'), 3, 0, ':') AS StartTime
FROM msdb.dbo.sysjobs j
INNER JOIN msdb.dbo.sysjobschedules js ON j.job_id = js.job_id
INNER JOIN msdb.dbo.sysschedules s ON js.schedule_id = s.schedule_id

-- Drop all foreign keys referencing Orders table
DECLARE @sql NVARCHAR(MAX);
DECLARE @fkName NVARCHAR(128);
DECLARE @parentTable NVARCHAR(128);

DECLARE fk_cursor CURSOR FOR
SELECT fk.name, OBJECT_NAME(fk.parent_object_id) AS parent_table
FROM sys.foreign_keys fk
WHERE fk.referenced_object_id = OBJECT_ID('Orders');

OPEN fk_cursor;
FETCH NEXT FROM fk_cursor INTO @fkName, @parentTable;

WHILE @@FETCH_STATUS = 0
BEGIN
    SET @sql = 'ALTER TABLE [dbo].[' + @parentTable + '] DROP CONSTRAINT [' + @fkName + ']';
    EXEC sp_executesql @sql;
    PRINT '    - Dropped FK: ' + @fkName;
    FETCH NEXT FROM fk_cursor INTO @fkName, @parentTable;
END

CLOSE fk_cursor;
DEALLOCATE fk_cursor;
GO

-- Drop PK on Orders
DECLARE @PKName NVARCHAR(128);
SELECT @PKName = kc.name
FROM sys.key_constraints kc
WHERE kc.parent_object_id = OBJECT_ID('Orders')
  AND kc.type = 'PK';

IF @PKName IS NOT NULL
BEGIN
    EXEC('ALTER TABLE [dbo].[Orders] DROP CONSTRAINT [' + @PKName + ']');
    PRINT '    - Primary Key dropped: ' + @PKName;
END
GO

-- Create new PK on ORDERS_FG

WITH (PAD_INDEX = OFF, STATISTICS_NORECOMPUTE = OFF, IGNORE_DUP_KEY = OFF, 
      ALLOW_ROW_LOCKS = ON, ALLOW_PAGE_LOCKS = ON)
ON [ORDERS_FG];

---------------------------------------------------------------
-- Trigger: Automatic status update when order is fully paid
---------------------------------------------------------------
CREATE TRIGGER trg_AutoCompleteOrder
ON dbo.CashRegister
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE o
    SET o.StatusID = 3   -- Completed
    FROM Orders o
    INNER JOIN inserted i ON o.OrderID = i.OrderID
    INNER JOIN OrderCosts oc ON o.OrderID = oc.OrderID
    WHERE (oc.BaseCost + oc.ComplicationCost + oc.FabricCost + oc.UrgencyCost) <=
          (SELECT SUM(Amount) FROM CashRegister WHERE OrderID = o.OrderID);
END;
GO

---------------------------------------------------------------
-- Trigger: Prevent deleting customers with active orders
---------------------------------------------------------------
CREATE TRIGGER trg_PreventCustomerDelete
ON dbo.Customers
INSTEAD OF DELETE
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1
        FROM deleted d
        JOIN Orders o ON d.CustomerID = o.CustomerID
        WHERE o.StatusID IN (1, 2) -- Active statuses
    )
    BEGIN
        RAISERROR('Cannot delete customer with active orders!', 16, 1);
        RETURN;
    END;

    DELETE FROM Customers
    WHERE CustomerID IN (SELECT CustomerID FROM deleted);
END;
GO

---------------------------------------------------------------
-- Trigger: Automatically fill completion date
---------------------------------------------------------------
CREATE TRIGGER trg_SetCompletionDate
ON dbo.Orders
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE o
    SET CompletionDate = GETDATE()
    FROM Orders o
    INNER JOIN inserted i ON o.OrderID = i.OrderID
    INNER JOIN deleted d ON o.OrderID = d.OrderID
    WHERE i.StatusID = 3      -- Completed
      AND d.StatusID <> 3
      AND o.CompletionDate IS NULL;
END;
GO

---------------------------------------------------------------
-- Trigger: Orders audit log
---------------------------------------------------------------
CREATE TRIGGER trg_Orders_Audit
ON dbo.Orders
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO dbo.OrderLogs (OrderID, OperationType, OperationDate, UserName)
    SELECT 
        COALESCE(i.OrderID, d.OrderID),
        CASE 
            WHEN i.OrderID IS NOT NULL AND d.OrderID IS NULL THEN 'INSERT'
            WHEN i.OrderID IS NOT NULL AND d.OrderID IS NOT NULL THEN 'UPDATE'
            WHEN i.OrderID IS NULL AND d.OrderID IS NOT NULL THEN 'DELETE'
        END,
        GETDATE(),
        ORIGINAL_LOGIN()
    FROM inserted i
    FULL JOIN deleted d ON i.OrderID = d.OrderID;
END;
GO

---------------------------------------------------------------
-- Trigger: Check fabric stock before fabric allocation
---------------------------------------------------------------
CREATE TRIGGER trg_CheckFabricStock
ON dbo.OrderFabrics
INSTEAD OF INSERT
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1
        FROM inserted i
        JOIN Fabrics f ON i.FabricID = f.FabricID
        WHERE i.FabricQuantity > f.Quantity
    )
    BEGIN
        RAISERROR('Not enough fabric in stock!', 16, 1);
        RETURN;
    END;

    UPDATE f
    SET f.Quantity = f.Quantity - i.FabricQuantity
    FROM Fabrics f
    JOIN inserted i ON f.FabricID = i.FabricID;

    INSERT INTO OrderFabrics (OrderID, FabricID, FabricQuantity)
    SELECT OrderID, FabricID, FabricQuantity
    FROM inserted;
END;
GO

CREATE TRIGGER trg_PreventCompleteUnpaidOrder
ON dbo.Orders
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1
        FROM inserted i
        JOIN deleted d ON i.OrderID = d.OrderID
        WHERE i.StatusID = 3 
          AND d.StatusID <> 3
          AND dbo.fn_GetOrderTotalCost(i.OrderID) > 
              (SELECT ISNULL(SUM(Amount), 0) 
               FROM CashRegister 
               WHERE OrderID = i.OrderID)
    )
    BEGIN
        RAISERROR('Cannot complete unpaid order!', 16, 1);
        ROLLBACK TRANSACTION;
    END
END;
GO

-- View: Revenue report

CREATE VIEW vw_RevenueReport
AS
SELECT 
    YEAR(cr.PaymentDate) AS Year,
    MONTH(cr.PaymentDate) AS Month,
    DATENAME(MONTH, cr.PaymentDate) AS MonthName,
    COUNT(DISTINCT cr.OrderID) AS OrdersCount,
    SUM(cr.Amount) AS TotalRevenue,
    AVG(cr.Amount) AS AvgOrderAmount
FROM CashRegister cr
GROUP BY YEAR(cr.PaymentDate), MONTH(cr.PaymentDate), DATENAME(MONTH, cr.PaymentDate);
GO

---------------------------------------------------------------
-- View: Tailor workload
---------------------------------------------------------------
CREATE VIEW vw_TailorWorkload
AS
SELECT 
    t.TailorID,
    t.FullName,
    c.CategoryName,
    COUNT(o.OrderID) AS ActiveOrders,
    SUM(CASE WHEN o.StatusID = 3 THEN 1 ELSE 0 END) AS CompletedOrders,
    AVG(DATEDIFF(DAY, o.OrderDate, COALESCE(o.CompletionDate, GETDATE()))) AS AvgDaysToComplete
FROM Tailors t
LEFT JOIN Orders o ON t.TailorID = o.TailorID
LEFT JOIN Categories c ON t.CategoryID = c.CategoryID
GROUP BY t.TailorID, t.FullName, c.CategoryName;
GO

---------------------------------------------------------------
-- View: Fabric inventory with warnings
---------------------------------------------------------------
CREATE VIEW vw_FabricInventory
AS
SELECT 
    f.FabricID,
    f.FabricName,
    ft.FabricTypeName,
    f.Quantity,
    f.Unit,
    f.Price,
    CASE 
        WHEN f.Quantity < 10 THEN 'Critical stock'
        WHEN f.Quantity < 50 THEN 'Low stock'
        ELSE 'Normal stock'
    END AS StockStatus
FROM Fabrics f
JOIN FabricTypes ft ON f.FabricTypeID = ft.FabricTypeID;
GO

---------------------------------------------------------------
-- View: Detailed order information
---------------------------------------------------------------
CREATE VIEW vw_OrderDetails
AS
SELECT 
    o.OrderID,
    o.OrderDate,
    c.CustomerName,
    c.Phone,
    t.FullName AS TailorName,
    p.ProductName,
    os.StatusName,
    oc.BaseCost,
    oc.ComplicationCost,
    oc.FabricCost,
    oc.UrgencyCost,
    (oc.BaseCost + oc.ComplicationCost + oc.FabricCost + oc.UrgencyCost) AS TotalCost,
    ISNULL(SUM(cr.Amount), 0) AS PaidAmount,
    (oc.BaseCost + oc.ComplicationCost + oc.FabricCost + oc.UrgencyCost) - 
        ISNULL(SUM(cr.Amount), 0) AS RemainingAmount
FROM Orders o
LEFT JOIN Customers c ON o.CustomerID = c.CustomerID
LEFT JOIN Tailors t ON o.TailorID = t.TailorID
LEFT JOIN Products p ON o.ProductID = p.ProductID
LEFT JOIN OrderStatuses os ON o.StatusID = os.StatusID
LEFT JOIN OrderCosts oc ON o.OrderID = oc.OrderID
LEFT JOIN CashRegister cr ON o.OrderID = cr.OrderID
GROUP BY o.OrderID, o.OrderDate, c.CustomerName, c.Phone, t.FullName,
         p.ProductName, os.StatusName, oc.BaseCost, oc.ComplicationCost,
         oc.FabricCost, oc.UrgencyCost;
GO

---------------------------------------------------------------
-- Procedure: Create basic order
---------------------------------------------------------------
CREATE PROCEDURE sp_CreateOrder
    @OrderDate DATE,
    @TailorID INT,
    @ProductID INT,
    @CustomerID INT,
    @UrgencyPercentage DECIMAL(5,2),
    @Language NVARCHAR(50),
    @StatusID INT
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM Tailors WHERE TailorID = @TailorID)
    BEGIN
        RAISERROR('Tailor not found!', 16, 1);
        RETURN;
    END;

    DECLARE @NewOrderID INT;
    SELECT @NewOrderID = ISNULL(MAX(OrderID), 0) + 1 FROM Orders;

    INSERT INTO Orders (OrderID, OrderDate, TailorID, ProductID, CustomerID, 
                        UrgencyPercentage, Language, StatusID)
    VALUES (@NewOrderID, @OrderDate, @TailorID, @ProductID, @CustomerID,
            @UrgencyPercentage, @Language, @StatusID);

    SELECT @NewOrderID AS CreatedOrderID;
END;
GO

---------------------------------------------------------------
-- Procedure: Calculate cost
---------------------------------------------------------------
CREATE PROCEDURE sp_CalculateOrderCost
    @OrderID INT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE 
        @BaseCost DECIMAL(10,2),
        @CompCost DECIMAL(10,2),
        @FabricCost DECIMAL(10,2),
        @UrgencyCost DECIMAL(10,2),
        @UrgencyPercent DECIMAL(5,2);

    SELECT @BaseCost = p.BasePrice
    FROM Orders o
    JOIN Products p ON o.ProductID = p.ProductID
    WHERE o.OrderID = @OrderID;

    SELECT @CompCost = ISNULL(SUM(c.PricePerElement * oc.ElementCount), 0)
    FROM OrderComplications oc
    JOIN Complications c ON oc.ComplicationID = c.ComplicationID
    WHERE oc.OrderID = @OrderID;

    SELECT @FabricCost = ISNULL(SUM(f.Price * ofb.FabricQuantity), 0)
    FROM OrderFabrics ofb
    JOIN Fabrics f ON ofb.FabricID = f.FabricID
    WHERE ofb.OrderID = @OrderID;

    SELECT @UrgencyPercent = UrgencyPercentage
    FROM Orders
    WHERE OrderID = @OrderID;

    SET @UrgencyCost = (@BaseCost + @CompCost + @FabricCost) * (@UrgencyPercent / 100.0);

    IF EXISTS (SELECT 1 FROM OrderCosts WHERE OrderID = @OrderID)
    BEGIN
        UPDATE OrderCosts
        SET BaseCost = @BaseCost,
            ComplicationCost = @CompCost,
            FabricCost = @FabricCost,
            UrgencyCost = @UrgencyCost
        WHERE OrderID = @OrderID;
    END
    ELSE
    BEGIN
        INSERT INTO OrderCosts (OrderID, BaseCost, ComplicationCost, FabricCost, UrgencyCost)
        VALUES (@OrderID, @BaseCost, @CompCost, @FabricCost, @UrgencyCost);
    END;
END;
GO

---------------------------------------------------------------
-- Procedure: Add a payment with validations
---------------------------------------------------------------
CREATE PROCEDURE sp_AddPayment
    @OrderID INT,
    @PaymentDate DATE,
    @Amount DECIMAL(10,2)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        IF NOT EXISTS (SELECT 1 FROM Orders WHERE OrderID = @OrderID)
        BEGIN
            RAISERROR('Order not found!', 16, 1);
            RETURN;
        END;

        DECLARE @TotalCost DECIMAL(10,2);
        SELECT @TotalCost = BaseCost + ComplicationCost + FabricCost + UrgencyCost
        FROM OrderCosts
        WHERE OrderID = @OrderID;

        DECLARE @PaidAmount DECIMAL(10,2);
        SELECT @PaidAmount = ISNULL(SUM(Amount), 0)
        FROM CashRegister
        WHERE OrderID = @OrderID;

        IF (@PaidAmount + @Amount) > @TotalCost
        BEGIN
            RAISERROR('Payment exceeds order total!', 16, 1);
            RETURN;
        END;

        INSERT INTO CashRegister (OrderID, PaymentDate, Amount)
        VALUES (@OrderID, @PaymentDate, @Amount);

        COMMIT TRANSACTION;

        SELECT 'Payment successfully added' AS Message,
               @Amount AS PaymentAmount,
               @TotalCost AS TotalCost,
               (@PaidAmount + @Amount) AS TotalPaid,
               (@TotalCost - @PaidAmount - @Amount) AS RemainingAmount;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
GO

---------------------------------------------------------------
-- Procedure: Tailor performance report
---------------------------------------------------------------
CREATE PROCEDURE sp_TailorReport
    @TailorID INT,
    @StartDate DATE = NULL,
    @EndDate DATE = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF @StartDate IS NULL SET @StartDate = DATEADD(MONTH, -1, GETDATE());
    IF @EndDate IS NULL SET @EndDate = GETDATE();

    SELECT 
        t.FullName AS TailorName,
        COUNT(o.OrderID) AS TotalOrders,
        SUM(CASE WHEN o.StatusID = 3 THEN 1 ELSE 0 END) AS CompletedOrders,
        SUM(CASE WHEN o.StatusID IN (1,2) THEN 1 ELSE 0 END) AS ActiveOrders,
        SUM(oc.BaseCost + oc.ComplicationCost + oc.FabricCost + oc.UrgencyCost) AS TotalRevenue,
        AVG(DATEDIFF(DAY, o.OrderDate, COALESCE(o.CompletionDate, GETDATE()))) AS AvgDaysToComplete
    FROM Tailors t
    LEFT JOIN Orders o ON t.TailorID = o.TailorID 
        AND o.OrderDate BETWEEN @StartDate AND @EndDate
    LEFT JOIN OrderCosts oc ON o.OrderID = oc.OrderID
    WHERE t.TailorID = @TailorID
    GROUP BY t.FullName;
END;
GO

---------------------------------------------------------------
-- Procedure: Find available tailors
---------------------------------------------------------------
CREATE PROCEDURE sp_FindAvailableTailors
    @CategoryID INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        t.TailorID,
        t.FullName,
        c.CategoryName,
        COUNT(o.OrderID) AS CurrentOrders,
        CASE 
            WHEN COUNT(o.OrderID) = 0 THEN 'Free'
            WHEN COUNT(o.OrderID) <= 3 THEN 'Available'
            ELSE 'Busy'
        END AS AvailabilityStatus
    FROM Tailors t
    LEFT JOIN Categories c ON t.CategoryID = c.CategoryID
    LEFT JOIN Orders o ON t.TailorID = o.TailorID 
        AND o.StatusID IN (1, 2)
    WHERE (@CategoryID IS NULL OR t.CategoryID = @CategoryID)
    GROUP BY t.TailorID, t.FullName, c.CategoryName
    HAVING COUNT(o.OrderID) < 5
    ORDER BY COUNT(o.OrderID);
END;
GO

---------------------------------------------------------------
-- Procedure: Full-order creation with fabric & complications
---------------------------------------------------------------
CREATE PROCEDURE sp_CreateFullOrder
    @OrderDate DATE,
    @TailorID INT,
    @ProductID INT,
    @CustomerID INT,
    @UrgencyPercentage DECIMAL(5,2),
    @Language NVARCHAR(50),
    @StatusID INT,
    @FabricList NVARCHAR(MAX),
    @ComplicationList NVARCHAR(MAX) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @NewOrderID INT;

        EXEC sp_CreateOrder 
            @OrderDate, @TailorID, @ProductID, @CustomerID,
            @UrgencyPercentage, @Language, @StatusID;

        SELECT @NewOrderID = MAX(OrderID) FROM Orders;

        INSERT INTO OrderFabrics (OrderID, FabricID, FabricQuantity)
        SELECT 
            @NewOrderID,
            JSON_VALUE(value, '$.FabricID'),
            JSON_VALUE(value, '$.Quantity')
        FROM OPENJSON(@FabricList);

        IF @ComplicationList IS NOT NULL
        BEGIN
            INSERT INTO OrderComplications (OrderID, ComplicationID, ElementCount)
            SELECT 
                @NewOrderID,
                JSON_VALUE(value, '$.ComplicationID'),
                JSON_VALUE(value, '$.ElementCount')
            FROM OPENJSON(@ComplicationList);
        END;

        EXEC sp_CalculateOrderCost @NewOrderID;

        COMMIT TRANSACTION;

        SELECT 
            @NewOrderID AS OrderID,
            dbo.fn_GetOrderTotalCost(@NewOrderID) AS TotalCost,
            'Order successfully created' AS Message;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH;
END;
GO

---------------------------------------------------------------
-- Function: Calculate total order cost
---------------------------------------------------------------
CREATE FUNCTION fn_GetOrderTotalCost
(
    @OrderID INT
)
RETURNS DECIMAL(10,2)
AS
BEGIN
    DECLARE @TotalCost DECIMAL(10,2);

    SELECT @TotalCost = BaseCost + ComplicationCost + FabricCost + UrgencyCost
    FROM OrderCosts
    WHERE OrderID = @OrderID;

    RETURN ISNULL(@TotalCost, 0);
END;
GO

---------------------------------------------------------------
-- Function: Get payment status
---------------------------------------------------------------
CREATE FUNCTION fn_GetPaymentStatus
(
    @OrderID INT
)
RETURNS NVARCHAR(50)
AS
BEGIN
    DECLARE @Status NVARCHAR(50);
    DECLARE @TotalCost DECIMAL(10,2);
    DECLARE @PaidAmount DECIMAL(10,2);

    SELECT @TotalCost = dbo.fn_GetOrderTotalCost(@OrderID);

    SELECT @PaidAmount = ISNULL(SUM(Amount), 0)
    FROM CashRegister
    WHERE OrderID = @OrderID;

    SET @Status = CASE
        WHEN @PaidAmount = 0 THEN 'Not paid'
        WHEN @PaidAmount < @TotalCost THEN 'Partially paid'
        WHEN @PaidAmount >= @TotalCost THEN 'Fully paid'
        ELSE 'Unknown'
    END;

    RETURN @Status;
END;
GO

---------------------------------------------------------------
-- Function: Customer order history (table function)
---------------------------------------------------------------
CREATE FUNCTION fn_GetCustomerOrderHistory
(
    @CustomerID INT
)
RETURNS TABLE
AS
RETURN
(
    SELECT 
        o.OrderID,
        o.OrderDate,
        p.ProductName,
        os.StatusName,
        dbo.fn_GetOrderTotalCost(o.OrderID) AS TotalCost,
        dbo.fn_GetPaymentStatus(o.OrderID) AS PaymentStatus
    FROM Orders o
    JOIN Products p ON o.ProductID = p.ProductID
    JOIN OrderStatuses os ON o.StatusID = os.StatusID
    WHERE o.CustomerID = @CustomerID
);
GO

---------------------------------------------------------------
-- Function: Check tailor availability (busy/not busy)
---------------------------------------------------------------
CREATE FUNCTION fn_CheckTailorAvailability
(
    @TailorID INT
)
RETURNS BIT
AS
BEGIN
    DECLARE @IsBusy BIT = 0;

    IF EXISTS (
        SELECT 1
        FROM Orders
        WHERE TailorID = @TailorID
          AND StatusID <> 3
    )
        SET @IsBusy = 1;

    RETURN @IsBusy;
END;
GO
